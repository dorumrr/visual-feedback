# Recipe: an S3-compatible bucket (S3, R2, Railway Buckets)

Each report becomes two objects in a private bucket, named and shaped as in the local recipe
(docs/recipes/local.md): `feedback/<id>/report.json`, where each screenshot's `image` is its file
name, and `feedback/<id>/screenshot-<n>.png` or `.webp` for each screenshot, numbered from 0.

The bucket stays private: nothing is public, and people read reports only through your own route,
behind your own admin check. `report.context` comes from the browser (docs/API.md), so an admin view
must not show it as proof of who sent a report. The images show the screen as the user saw it, with nothing masked
(SPEC §7).

## 1. Create the bucket, install, configure

Create the bucket with public access off (on S3, keep Block Public Access on), then check that an
object URL opened without a key is refused.

```sh
npm install aws4fetch
```

Set these on the server. Never send them to the browser.

| variable | for example |
|---|---|
| `S3_ENDPOINT` | `https://s3.eu-west-2.amazonaws.com`, `https://<account>.r2.cloudflarestorage.com`, or your Railway Bucket's endpoint |
| `S3_REGION` | `eu-west-2`; optional, `auto` when unset, which R2 takes and AWS does not |
| `S3_BUCKET` | `feedback` |
| `S3_ACCESS_KEY_ID`, `S3_SECRET_ACCESS_KEY` | a key that may read and write this bucket only |

## 2. The route

The widget posts to it with `createFeedback({ endpoint: "/api/feedback" })`. This is a Next.js
App Router route. As written, anyone who can reach `POST` can write to the bucket: before production,
put it behind your own session check and a rate limit. A report already in the bucket is never
replaced: the route checks for it first, and every write is create-only (`If-None-Match: *`, which
S3 and R2 support). On a store that ignores that header, only the first check protects it.

```ts
// app/api/feedback/route.ts
import { AwsClient } from "aws4fetch";
import { FeedbackRequestError, parseFeedbackRequest } from "visual-feedback/server";

// Replace with your own admin check, from your session. Until then, nobody can read.
async function canRead(_request: Request): Promise<boolean> {
  return false;
}

function bucket() {
  const { S3_ENDPOINT, S3_REGION, S3_BUCKET, S3_ACCESS_KEY_ID, S3_SECRET_ACCESS_KEY } = process.env;
  if (!S3_ENDPOINT || !S3_BUCKET || !S3_ACCESS_KEY_ID || !S3_SECRET_ACCESS_KEY) {
    throw new Error("Set S3_ENDPOINT, S3_BUCKET, S3_ACCESS_KEY_ID and S3_SECRET_ACCESS_KEY.");
  }
  const client = new AwsClient({
    accessKeyId: S3_ACCESS_KEY_ID,
    secretAccessKey: S3_SECRET_ACCESS_KEY,
    region: S3_REGION || "auto",
    service: "s3",
    // aws4fetch retries 5xx about 50 s by default; the user would wait on "Sending…" that long.
    retries: 2,
  });
  const base = `${S3_ENDPOINT.replace(/\/+$/, "")}/${S3_BUCKET}/feedback`;
  return { client, url: (id: string, name: string) => `${base}/${id}/${name}` };
}

export async function POST(request: Request): Promise<Response> {
  try {
    const { report } = await parseFeedbackRequest(request);
    const { client, url } = bucket();
    // A report already stored under this id stays: a retry, or a reused id, changes nothing.
    const stored = await client.fetch(url(report.id, "report.json"), { method: "HEAD" });
    if (stored.ok) return new Response(null, { status: 204 });
    // S3 answers 403 for a missing key when the key may not list; any other answer is unclear.
    if (stored.status !== 403 && stored.status !== 404) {
      throw new Error(`The bucket answered ${stored.status} for report.json.`);
    }
    const name = (image: Blob, i: number) =>
      `screenshot-${i}.${image.type === "image/png" ? "png" : "webp"}`;
    const screenshots = report.screenshots.map((shot, i) => ({ ...shot, image: name(shot.image, i) }));
    // The images first: a report.json in the bucket then always has its images beside it.
    const puts: [string, BodyInit, string][] = [
      ...report.screenshots.map(
        ({ image }, i): [string, BodyInit, string] => [name(image, i), image, image.type],
      ),
      ["report.json", JSON.stringify({ ...report, screenshots }), "application/json"],
    ];
    for (const [file, body, type] of puts) {
      const put = await client.fetch(url(report.id, file), {
        method: "PUT",
        body,
        // Create-only: S3 and R2 answer 412 and keep the object a request stored first.
        headers: { "content-type": type, "if-none-match": "*" },
      });
      if (!put.ok && put.status !== 412) {
        throw new Error(`The bucket answered ${put.status} for ${file}.`);
      }
    }
    return new Response(null, { status: 204 });
  } catch (error) {
    if (error instanceof FeedbackRequestError) {
      return new Response(error.message, { status: error.status });
    }
    throw error;
  }
}

// GET /api/feedback?id=<report id>&file=report.json | screenshot-<n>.png | screenshot-<n>.webp
export async function GET(request: Request): Promise<Response> {
  if (!(await canRead(request))) return new Response(null, { status: 403 });
  const query = new URL(request.url).searchParams;
  const id = query.get("id") ?? "";
  const file = query.get("file") ?? "";
  // Only a report id and a known file name, so the key cannot leave feedback/<id>/.
  const known = file === "report.json" || /^screenshot-\d\.(png|webp)$/.test(file);
  if (!/^[0-9a-f-]{36}$/.test(id) || !known) {
    return new Response(null, { status: 400 });
  }
  const { client, url } = bucket();
  const object = await client.fetch(url(id, file));
  // S3 answers 403, not 404, for a missing key when the key may not list the bucket.
  const missing = object.status === 403 || object.status === 404;
  if (!object.ok) return new Response(null, { status: missing ? 404 : 502 });
  return new Response(object.body, {
    headers: {
      "content-type": object.headers.get("content-type") ?? "application/octet-stream",
      "cache-control": "private, no-store",
    },
  });
}
```
